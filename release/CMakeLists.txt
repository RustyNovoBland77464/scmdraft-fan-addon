cmake_minimum_required(VERSION 3.16)
project(ScmDraftFanAddon VERSION 0.1.1 LANGUAGES C CXX)
enable_language(RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
if(NOT MINGW OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Use 32-bit MinGW with third_party/mingw32.cmake. Native macOS packaging is separate.")
endif()
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CASC_BUILD_SHARED_LIB OFF CACHE BOOL "" FORCE)
set(CASC_BUILD_STATIC_LIB ON CACHE BOOL "" FORCE)
set(CASC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CASC_UNICODE OFF CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_ZLIB TRUE)
add_subdirectory("${ROOT}/third_party/CascLib" "${CMAKE_BINARY_DIR}/CascLib" EXCLUDE_FROM_ALL)
add_subdirectory("${ROOT}/third_party/minhook" "${CMAKE_BINARY_DIR}/minhook" EXCLUDE_FROM_ALL)
add_compile_options(-O2 -pthread "-ffile-prefix-map=${ROOT}=." -Wno-multichar)
add_link_options(-static -static-libgcc -static-libstdc++ -pthread)
set(BW "${ROOT}/third_party/openbw")
add_library(openbw_live OBJECT "${ROOT}/hd/src/live.cpp" "${ROOT}/hd/src/openbw_stubs.cpp")
target_include_directories(openbw_live PRIVATE "${BW}" "${BW}/deps" "${BW}/ui")
target_compile_options(openbw_live PRIVATE -O3 -w)
add_library(scmhd SHARED
  "${ROOT}/hd/src/scmhd.cpp" "${ROOT}/hd/src/hd_sprites.cpp"
  "${ROOT}/hd/src/symmetry.cpp" "${ROOT}/hd/src/mining.cpp"
  "${ROOT}/hd/src/scmhd.def" $<TARGET_OBJECTS:openbw_live>)
set_target_properties(scmhd PROPERTIES PREFIX "" SUFFIX ".sdp")
target_include_directories(scmhd PRIVATE "${ROOT}/third_party/CascLib/src")
target_link_libraries(scmhd PRIVATE casc_static minhook wininet ws2_32 gdi32 user32 advapi32 msimg32)
target_link_options(scmhd PRIVATE -Wl,--enable-stdcall-fixup)
add_executable(minesim "${ROOT}/minesim/minesim.cpp")
target_include_directories(minesim PRIVATE "${BW}" "${BW}/deps" "${ROOT}/third_party/CascLib/src")
target_compile_options(minesim PRIVATE -w)
target_link_libraries(minesim PRIVATE casc_static ws2_32 wininet)
set(PAYLOAD_DIRECTORY "${CMAKE_BINARY_DIR}")
set(NOTICES "${CMAKE_BINARY_DIR}/ScmDraft-Fan-Addon-NOTICES.txt")
file(WRITE "${NOTICES}" "ScmDraft Fan Add-on - third-party notices\n\n")
foreach(NOTICE IN ITEMS release/docs/LICENSE release/docs/THIRD_PARTY_NOTICES.md
    third_party/CascLib/LICENSE third_party/minhook/LICENSE.txt
    third_party/openbw/deps/asio/LICENSE_1_0.txt third_party/CascLib/src/zlib/zlib.h
    third_party/CascLib/src/hashes/sha1.h third_party/CascLib/src/hashes/md5.h
    third_party/CascLib/src/overwatch/aes.h third_party/CascLib/src/jenkins/lookup.h)
  file(READ "${ROOT}/${NOTICE}" CONTENT)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/${NOTICE}")
  file(APPEND "${NOTICES}" "\n--- ${NOTICE} ---\n${CONTENT}\n")
endforeach()
configure_file(installer/payload.rc.in "${CMAKE_BINARY_DIR}/payload.rc" @ONLY)
set_source_files_properties("${CMAKE_BINARY_DIR}/payload.rc" PROPERTIES OBJECT_DEPENDS "${CMAKE_BINARY_DIR}/scmhd.sdp;${CMAKE_BINARY_DIR}/minesim.exe;${NOTICES}")
add_executable(fan-addon-setup WIN32 installer/setup.cpp installer/setup.rc "${CMAKE_BINARY_DIR}/payload.rc")
add_dependencies(fan-addon-setup scmhd minesim)
target_include_directories(fan-addon-setup PRIVATE installer)
target_compile_definitions(fan-addon-setup PRIVATE UNICODE _UNICODE)
target_link_options(fan-addon-setup PRIVATE -municode)
target_link_libraries(fan-addon-setup PRIVATE comdlg32 advapi32 shell32 ole32)
add_executable(installer-check installer/check.cpp)
target_link_libraries(installer-check PRIVATE advapi32)
